% --- Native Hebrew (RTL) support ---
% The PDF build uses XeLaTeX (required by mainfont). pandoc already loads
% babel because lang: en is set; we add Hebrew to that same babel instance,
% which pulls in correct RTL/bidi handling. Requires Ezra SIL (SIL, OFL).
%   Install: download from https://software.sil.org/ezra/ and copy the .ttf
%   into ~/Library/Fonts/ (macOS) or ~/.fonts/ (Linux).
% Use in text:  \foreignlanguage{hebrew}{בְּרֵאשִׁית}
% Polytonic Greek needs no setup — it is covered by the main font (Gentium Plus).
\babelprovide[import]{hebrew}
\babelfont[hebrew]{rm}{Ezra SIL}
